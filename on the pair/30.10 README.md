using System.Data;
using Dapper;
using Npgsql;

var builder = WebApplication.CreateBuilder(args);
var app = builder.Build();


string cs = "Host=localhost;Port=5432;Database=pharmacy;Username=postgres;Password=123";

app.MapGet("/", () => "API Аптеки работает! Используйте /meds, /sale, /meds/report");

app.MapGet("/meds", async (string? q) =>
{
    using var conn = new NpgsqlConnection(cs);
    var sql = @"
        SELECT * FROM meds WHERE name ILIKE @Q OR @Q IS NULL;
        SELECT COUNT(*) FROM meds WHERE name ILIKE @Q OR @Q IS NULL;
    ";
    
    using var multi = await conn.QueryMultipleAsync(sql, new { Q = $"%{q}%" });
    var meds = await multi.ReadAsync<dynamic>();
    var total = await multi.ReadSingleAsync<int>();

    return Results.Ok(new { Total = total, Items = meds });
});

app.MapPost("/sale", async (SaleReq s) =>
{
    using var conn = new NpgsqlConnection(cs);
    await conn.OpenAsync();
    await using var tx = await conn.BeginTransactionAsync();

    try
    {
        var currentQty = await conn.ExecuteScalarAsync<int>(
            "SELECT qty FROM meds WHERE id = @Id FOR UPDATE", 
            new { Id = s.MedId }, transaction: tx);

        if (currentQty < s.Qty)
        {
            await tx.RollbackAsync();
            return Results.BadRequest("Недостаточно товара на складе.");
        }

        await conn.ExecuteAsync(
            "UPDATE meds SET qty = qty - @Qty WHERE id = @Id", 
            new { Qty = s.Qty, Id = s.MedId }, transaction: tx);

        await conn.ExecuteAsync(
            "INSERT INTO sales (med_id, qty) VALUES (@MedId, @Qty)", 
            new { MedId = s.MedId, Qty = s.Qty }, transaction: tx);

        await tx.CommitAsync();
        return Results.Ok(new { Message = "Продажа успешно оформлена." });
    }
    catch (Exception ex)
    {
        await tx.RollbackAsync();
        return Results.Problem(ex.Message);
    }
});

app.MapGet("/meds/report", () =>
{
    using var conn = new NpgsqlConnection(cs);
    conn.Open(); 

    string sql = @"
        SELECT m.name, m.price, COALESCE(SUM(s.qty), 0) AS sold_qty
        FROM meds m
        LEFT JOIN sales s ON m.id = s.med_id
        GROUP BY m.id, m.name, m.price
        ORDER BY sold_qty DESC;
    ";

    using var adapter = new NpgsqlDataAdapter(sql, conn);
    var ds = new DataSet();
    adapter.Fill(ds, "SalesReport");
    
    var table = ds.Tables["SalesReport"];
    var result = new List<Dictionary<string, object?>>();
    foreach (DataRow row in table.Rows)
    {
        var dict = new Dictionary<string, object?>();
        foreach (DataColumn col in table.Columns)
        {
            dict[col.ColumnName] = row[col];
        }
        result.Add(dict);
    }

    return Results.Ok(result);
});

app.Run();

record SaleReq(int MedId, int Qty);
